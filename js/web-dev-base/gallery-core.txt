gallery-core.js (web-dev-base)

normalizeImageEntry — string or {src, alt?} → {src, alt}

wrapGalleryIndex — circular index for prev/next

setGalleryBodyScrollLocked — true = hide body scroll (lightbox)

bindDocumentGalleryKeydown — Escape / arrows when open; returns dispose()

initGalleryLightboxFromDom — hooks .gallery-item + #lightbox when present

createDraggableGallery(root, options?) — strip + zoom; click slide opens, click same slide again closes; setImages, addImages, clear, destroy; zoomThumbFill true = fixed thumb height + cover crop in rail; default = variable thumb height, full rail width, whole image; zoomPaneMaxHeight = CSS string sets --draggable-gallery-zoom-pane-max (rail scrolls inside cap, zoom pane no scroll, both capped; default in CSS min(58vh,520px) if unset)
